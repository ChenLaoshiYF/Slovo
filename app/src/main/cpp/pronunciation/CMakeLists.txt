cmake_minimum_required(VERSION 3.22)
project(slovo_pronunciation LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option BUILD_ESPEAK_NG_TESTS BUILD_ESPEAK_NG_EXE USE_ASYNC USE_MBROLA USE_LIBSONIC USE_LIBPCAUDIO USE_KLATT USE_SPEECHPLAYER)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(EXTRA_cmn ON CACHE BOOL "" FORCE)
set(EXTRA_ru ON CACHE BOOL "" FORCE)
add_subdirectory("${ESPEAK_SOURCE}" espeak-build EXCLUDE_FROM_ALL)
# Android loads this library from the APK; no runtime search path is needed.
set_property(TARGET espeak-ng PROPERTY LINK_LIBRARIES "espeak-ng-config;ucd;m;espeak-include")
set_property(TARGET espeak-ng PROPERTY INTERFACE_LINK_LIBRARIES "ucd;m")
add_library(slovo_pronunciation SHARED pronunciation.cpp)
target_compile_features(slovo_pronunciation PRIVATE cxx_std_17)
target_include_directories(slovo_pronunciation PRIVATE "${ESPEAK_SOURCE}/src/include")
target_link_libraries(slovo_pronunciation PRIVATE espeak-ng)
target_link_options(slovo_pronunciation PRIVATE "-Wl,--exclude-libs,ALL")
